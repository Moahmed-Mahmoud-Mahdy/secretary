    28→// ============================================================
    29→// AI chat use case (BRD §5-§9, §42) — the orchestration heart:
    30→//   User Input → Intent Detection → Command Generation →
    31→//   Validation → Business Rules → Command Execution.
    32→//
    33→// The AI never touches the database: it only proposes actions;
    34→// this use case validates and executes them through the other
    35→// use cases (Smart Auto-Execute: creates run instantly,
    36→// destructive/updates need user confirmation).
    37→// ============================================================
    38→
    39→export interface ExecutedActionDTO {
    40→  type: 'TASK' | 'EVENT' | 'EXPENSE' | 'INCOME' | 'BUDGET' | 'PROJECT' | 'PLAN' | 'TASK_COMPLETED';
    41→  action: 'CREATED' | 'COMPLETED' | 'EXECUTED';
    42→  summary: string;
    43→  refId?: string;
    44→}
    45→
    46→export interface PendingActionDTO {
    47→  id: string;
    48→  type: 'DELETE_TASK' | 'UPDATE_TASK' | 'DELETE_EVENT';
    49→  title: string;
    50→  summary: string;
    51→  payload: Record<string, unknown>;
    52→}
    53→
    54→export interface QueryDataDTO {
    55→  queryType: string;
    56→  question: string;
    57→  data: Record<string, unknown>;
    58→}
    59→
    60→export interface ChatResultDTO {
    61→  reply: string;
    62→  intent: string;
    63→  executed: ExecutedActionDTO[];
    64→  pending: PendingActionDTO[];
    65→  query: QueryDataDTO | null;
    66→  failed: string[];
    67→}
    68→
    69→const MAX_ACTIONS_PER_MESSAGE = 8;
    70→
    71→/** Actions that always need explicit user confirmation (BRD §8). */
    72→const CONFIRMATION_REQUIRED = new Set(['DELETE_TASK', 'UPDATE_TASK', 'DELETE_EVENT']);

    92→  async handleMessage(userId: string, message: string): Promise<ChatResultDTO> {
    93→    const text = message.trim();
    94→    if (!text) throw new AiInterpretationError('اكتب حاجة الأول 😄');
    95→
    96→    const context = await this.buildContext(userId);
    97→    const interpretation = await this.ai.interpret({ message: text, context });
    98→    if (process.env.NODE_ENV !== 'production') {
    99→      console.log('[ai-chat] interpretation:', JSON.stringify(interpretation).slice(0, 500));
   100→    }
   101→
   102→    const executed: ExecutedActionDTO[] = [];
   103→    const pending: PendingActionDTO[] = [];
   104→    const failed: string[] = [];
   105→    let query: QueryDataDTO | null = null;
   106→    let chitchat = interpretation.intent === 'CHITCHAT';
   107→
   108→    const actions = (interpretation.actions ?? []).slice(0, MAX_ACTIONS_PER_MESSAGE);
   109→    // Safety net: when the model returns an intent without actions,
   110→    // synthesize the command from the intent itself.
   111→    const effectiveActions: AiAction[] =
   112→      actions.length > 0 ? actions : (synthesizeActionFromIntent(interpretation.intent, text) ?? []);
   113→    for (const action of effectiveActions) {
   114→      try {
   115→        await this.processAction(userId, action, executed, pending, (q) => {
   116→          query = q;
   117→        });
   118→      } catch (error) {
   119→        failed.push(actionLabel(action));
   120→        if (process.env.NODE_ENV !== 'production') {
   121→          console.error('[ai-chat] action failed:', action.type, error);
   122→        }
   123→      }
   124→      if ((action.type ?? '').toUpperCase() === 'CHITCHAT') chitchat = true;
   125→    }
   126→
   127→    let reply: string;
   128→    const queryValue = query as QueryDataDTO | null;
   129→    if (queryValue && queryValue.question) {
   130→      reply = await this.safeAnswerQuestion(queryValue, context.firstName);
   131→    } else if (executed.length > 0 || pending.length > 0 || failed.length > 0) {
   132→      reply = this.buildReply(context.firstName, executed, pending, failed);
   133→    } else if (chitchat) {
   134→      reply = await this.safeSmallTalk(text, context.firstName);
   135→    } else {
   136→      reply = `مش متأكد إني فهمت قصدك يا ${context.firstName} — ممكن تقولها لي بطريقة تانية؟`;
   137→    }
   138→
   139→    return { reply, intent: interpretation.intent ?? 'UNKNOWN', executed, pending, query: queryValue, failed };
   140→  }
   141→
   142→  /** Executes a previously-confirmed action (BRD §8 confirmation flow). */
   143→  async executeConfirmedAction(userId: string, pending: PendingActionDTO): Promise<ExecutedActionDTO> {
   144→    const executed: ExecutedActionDTO[] = [];
   145→    switch (pending.type) {
   146→      case 'DELETE_TASK': {
   147→        const taskId = String(payloadValue(pending.payload, 'taskId'));
   148→        const task = await this.tasks.findById(userId, taskId);
   149→        await this.taskUseCases.delete(userId, taskId);
   150→        return { type: 'TASK', action: 'EXECUTED', summary: `تمسحت مهمة «${task?.title ?? 'المهمة'}»`, refId: taskId };
   151→      }
   152→      case 'DELETE_EVENT': {
   153→        const eventId = String(payloadValue(pending.payload, 'eventId'));
   154→        const event = await this.events.findById(userId, eventId);
   155→        await this.events.delete(userId, eventId);
   156→        return { type: 'EVENT', action: 'EXECUTED', summary: `تمسح حدث «${event?.title ?? 'الحدث'}»`, refId: eventId };
   157→      }
   158→      case 'UPDATE_TASK': {
   159→        const taskId = String(payloadValue(pending.payload, 'taskId'));
   160→        const fields = (payloadValue(pending.payload, 'fields') ?? {}) as Record<string, unknown>;
   161→        const updated = await this.taskUseCases.update(userId, taskId, fields as never);
   162→        return { type: 'TASK', action: 'EXECUTED', summary: `عدّلت مهمة «${updated.title}»`, refId: taskId };
   163→      }
   164→      default:
   165→        throw new AiInterpretationError('نوع الطلب ده مش محتاج تأكيد');
   166→    }
   167→  }

   169→  // ---------------- internals ----------------
   170→
   171→  private async processAction(
   172→    userId: string,
   173→    action: AiAction,
   174→    executed: ExecutedActionDTO[],
   175→    pending: PendingActionDTO[],
   176→    setQuery: (q: QueryDataDTO) => void
   177→  ): Promise<void> {
   178→    const type = String(action.type ?? '').toUpperCase();
   179→
   180→    switch (type) {
   181→      case 'CREATE_TASK': {
   182→        const created = await this.taskUseCases.create(userId, {
   183→          title: str(action.title) ?? 'مهمة جديدة',
   184→          description: str(action.description),
   185→          priority: str(action.priority) ?? undefined,
   186→          estimatedMinutes: num(action.estimatedMinutes),
   187→          deadline: str(action.deadline),
   188→          projectId: await this.resolveProjectId(userId, str(action.projectName)),
   189→        });
   190→        executed.push({ type: 'TASK', action: 'CREATED', summary: this.taskSummary(created), refId: created.id });
   191→        return;
   192→      }
   193→      case 'CREATE_EVENT': {
   194→        const startAt = str(action.startAt);
   195→        if (!startAt) throw new Error('event without startAt');
   196→        const created = await this.eventUseCases.create(userId, {
   197→          title: str(action.title) ?? 'حدث',
   198→          startAt,
   199→          endAt: str(action.endAt) ?? null,
   200→          recurrence: str(action.recurrence),
   201→        });
   202→        executed.push({ type: 'EVENT', action: 'CREATED', summary: this.eventSummary(created.title, created.startAt), refId: created.id });
   203→        return;
   204→      }
   205→      case 'CREATE_EXPENSE': {
   206→        const amount = num(action.amount);
   207→        if (!amount) throw new Error('expense without amount');
   208→        const created = await this.financeUseCases.createExpense(userId, {
   209→          amount,
   210→          category: str(action.category) ?? undefined,
   211→          description: str(action.description),
   212→          date: str(action.date),
   213→        });
   214→        const categoryLabel = CATEGORY_LABELS_AR[created.category] ?? '';
   215→        const summary = `سجلت ${formatAmount(created.amount)} جنيه على ${created.description || categoryLabel}`;
   216→        executed.push({ type: 'EXPENSE', action: 'CREATED', summary, refId: created.id });
   217→        return;
   218→      }
   219→      case 'CREATE_INCOME': {
   220→        const amount = num(action.amount);
   221→        if (!amount) throw new Error('income without amount');
   222→        const created = await this.financeUseCases.createIncome(userId, {
   223→          amount,
   224→          source: str(action.source),
   225→          date: str(action.date),
   226→        });
   227→        executed.push({ type: 'INCOME', action: 'CREATED', summary: `سجلت دخل ${formatAmount(created.amount)} جنيه`, refId: created.id });
   228→        return;
   229→      }
   230→      case 'SET_BUDGET': {
   231→        const amount = num(action.amount);
   232→        if (!amount) throw new Error('budget without amount');
   233→        await this.financeUseCases.setBudget(userId, amount);
   234→        executed.push({ type: 'BUDGET', action: 'CREATED', summary: `ظبطت ميزانية الشهر على ${formatAmount(amount)} جنيه` });
   235→        return;
   236→      }
   237→      case 'SET_CATEGORY_BUDGET': {
   238→        const amount = num(action.amount);
   239→        const category = str(action.category) ?? 'OTHER';
   240→        if (!amount) throw new Error('category budget without amount');
   241→        await this.financeUseCases.setCategoryBudget(userId, category, amount);
   242→        const label = CATEGORY_LABELS_AR[category.toUpperCase() as keyof typeof CATEGORY_LABELS_AR] ?? category;
   243→        executed.push({
   244→          type: 'BUDGET',
   245→          action: 'CREATED',
   246→          summary: `ظبطت حد صرف ${label} على ${formatAmount(amount)} جنيه في الشهر`,
   247→        });
   248→        return;
   249→      }
   250→      case 'CREATE_PROJECT': {
   251→        const name = str(action.name) ?? str(action.title);
   252→        if (!name) throw new Error('project without name');
   253→        const created = await this.projectUseCases.create(userId, { name, deadline: str(action.deadline) });
   254→        executed.push({ type: 'PROJECT', action: 'CREATED', summary: `عملتلك مشروع «${created.name}»`, refId: created.id });
   255→        return;
   256→      }
   257→      case 'CREATE_PROJECT_WITH_TASKS': {
   258→        const name = str(action.name) ?? str(action.title);
   259→        if (!name) throw new Error('project breakdown without name');
   260→        const created = await this.projectUseCases.create(userId, {
   261→          name,
   262→          description: str(action.description),
   263→          deadline: str(action.deadline),
   264→        });
   265→        const rawTasks = Array.isArray(action.tasks) ? action.tasks.slice(0, 8) : [];
   266→        const titles: string[] = [];
   267→        for (const raw of rawTasks) {
   268→          const t = (raw ?? {}) as Record<string, unknown>;
   269→          const title = str(t.title);
   270→          if (!title) continue;
   271→          await this.taskUseCases.create(userId, {
   272→            title,
   273→            priority: str(t.priority) ?? 'MEDIUM',
   274→            estimatedMinutes: num(t.estimatedMinutes),
   275→            projectId: created.id,
   276→          });
   277→          titles.push(title);
   278→        }
   279→        executed.push({
   280→          type: 'PROJECT',
   281→          action: 'CREATED',
   282→          summary:
   283→            titles.length > 0
   284→              ? `عملتلك مشروع «${created.name}» فيه ${titles.length} مهام: ${titles.join('، ')}`
   285→              : `عملتلك مشروع «${created.name}»`,
   286→          refId: created.id,
   287→        });
   288→        return;
   289→      }
   290→      case 'ADD_SUBTASKS': {
   291→        const parentName = str(action.taskName) ?? str(action.title) ?? '';
   292→        const all = await this.tasks.listAll(userId);
   293→        const parent = matchTaskTitle(all, parentName);
   294→        if (!parent) {
   295→          executed.push({ type: 'TASK', action: 'EXECUTED', summary: `ملقيتش مهمة باسم «${parentName || '؟'}» أضيفلها خطوات` });
   296→          return;
   297→        }
   298→        const rawSubtasks = Array.isArray(action.subtasks) ? action.subtasks.slice(0, 10) : [];
   299→        let added = 0;
   300→        for (const raw of rawSubtasks) {
   301→          const title = typeof raw === 'string' ? raw.trim() : str((raw as Record<string, unknown>)?.title);
   302→          if (!title) continue;
   303→          await this.taskUseCases.create(userId, { title, parentId: parent.id, projectId: parent.projectId });
   304→          added += 1;
   305→        }
   306→        executed.push({
   307→          type: 'TASK',
   308→          action: 'CREATED',
   309→          summary:
   310→            added > 0
   311→              ? `ضفت ${added} ${added === 1 ? 'خطوة' : 'خطوات'} تحت «${parent.title}»`
   312→              : `مفيش خطوات أقدر أضيفها تحت «${parent.title}»`,
   313→          refId: parent.id,
   314→        });
   315→        return;
   316→      }
   317→      case 'COMPLETE_TASK': {
   318→        const all = await this.tasks.listAll(userId);
   319→        const task = matchTaskTitle(all, str(action.taskName) ?? str(action.title) ?? '');
   320→        if (!task) {
   321→          executed.push({ type: 'TASK', action: 'EXECUTED', summary: `ملقيتش مهمة باسم «${str(action.taskName) ?? '؟'}» عشان أقفلها` });
   322→          return;
   323→        }
   324→        await this.taskUseCases.update(userId, task.id, { status: 'COMPLETED' });
   325→        executed.push({ type: 'TASK_COMPLETED', action: 'COMPLETED', summary: `برافو! قفلت مهمة «${task.title}» ✅`, refId: task.id });
   326→        return;
   327→      }
   328→      case 'DELETE_TASK': {
   329→        const all = await this.tasks.listAll(userId);
   330→        const task = matchTaskTitle(all, str(action.taskName) ?? '');
   331→        if (!task) {
   332→          executed.push({ type: 'TASK', action: 'EXECUTED', summary: `ملقيتش مهمة باسم «${str(action.taskName) ?? '؟'}»` });
   333→          return;
   334→        }
   335→        this.pendingCounter += 1;
   336→        pending.push({
   337→          id: `p${this.pendingCounter}_${Date.now()}`,
   338→          type: 'DELETE_TASK',
   339→          title: task.title,
   340→          summary: `مسح مهمة «${task.title}»`,
   341→          payload: { taskId: task.id },
   342→        });
   343→        return;
   344→      }
   345→      case 'UPDATE_TASK': {
   346→        const all = await this.tasks.listAll(userId);
   347→        const task = matchTaskTitle(all, str(action.taskName) ?? '');
   348→        if (!task) {
   349→          executed.push({ type: 'TASK', action: 'EXECUTED', summary: `ملقيتش مهمة باسم «${str(action.taskName) ?? '؟'}»` });
   350→          return;
   351→        }
   352→        const fields = sanitizeTaskFields(action.fields);
   353→        this.pendingCounter += 1;
   354→        pending.push({
   355→          id: `p${this.pendingCounter}_${Date.now()}`,
   356→          type: 'UPDATE_TASK',
   357→          title: task.title,
   358→          summary: `تعديل مهمة «${task.title}»: ${describeFields(fields)}`,
   359→          payload: { taskId: task.id, fields },
   360→        });
   361→        return;
   362→      }
   363→      case 'DELETE_EVENT': {
   364→        const eventName = str(action.eventName) ?? '';
   365→        const event = await this.matchEvent(userId, eventName);
   366→        if (!event) {
   367→          executed.push({ type: 'EVENT', action: 'EXECUTED', summary: `ملقيتش حدث باسم «${eventName || '؟'}»` });
   368→          return;
   369→        }
   370→        this.pendingCounter += 1;
   371→        pending.push({
   372→          id: `p${this.pendingCounter}_${Date.now()}`,
   373→          type: 'DELETE_EVENT',
   374→          title: event.title,
   375→          summary: `مسح حدث «${event.title}»`,
   376→          payload: { eventId: event.id },
   377→        });
   378→        return;
   379→      }
   380→      case 'PLAN_DAY': {
   381→        const dateStr = str(action.date);
   382→        const plan = await this.planningUseCases.generatePlan(userId, dateStr ?? undefined);
   383→        executed.push({
   384→          type: 'PLAN',
   385→          action: 'EXECUTED',
   386→          summary: `ظبطتلك خطة ${dateStr ? relativeDayArabic(parseWallIso(dateStr), nowWall()) : 'النهارده'} — ${plan.slots.length} ${plan.slots.length === 1 ? 'مهمة مجدولة' : 'مهام مجدولة'}`,
   387→        });
   388→        setQuery({
   389→          queryType: 'PLAN_RESULT',
   390→          question: 'عرض الخطة',
   391→          data: { date: plan.date, slots: plan.slots, unplanned: plan.unplanned },
   392→        });
   393→        return;
   394→      }
   395→      case 'QUERY': {
   396→        const queryType = (str(action.queryType) ?? 'GENERAL').toUpperCase();
   397→        const question = str(action.question) ?? str(action.raw) ?? 'سؤال المستخدم';
   398→        const data = await this.fetchQueryData(userId, queryType);
   399→        setQuery({ queryType, question, data });
   400→        return;
   401→      }
   402→      case 'CHITCHAT':
   403→        return; // handled after the loop
   404→      default:
   405→        throw new Error(`unknown action type: ${type}`);
   406→    }
   407→  }

   409→  private async fetchQueryData(userId: string, queryType: string): Promise<Record<string, unknown>> {
   410→    const now = nowWall();
   411→    switch (queryType) {
   412→      case 'FINANCE_SUMMARY': {
   413→        const s = await this.financeUseCases.summary(userId);
   414→        return {
   415→          monthSpent: s.monthSpent,
   416→          budget: s.budget,
   417→          remaining: s.remaining,
   418→          spentToday: s.spentToday,
   419→          incomeThisMonth: s.incomeThisMonth,
   420→          dailyAverage: Math.round(s.dailyAverage),
   421→          byCategory: s.byCategory.slice(0, 5),
   422→          categoryLimits: s.categoryLimits,
   423→        };
   424→      }
   425→      case 'BUDGET_STATUS': {
   426→        const s = await this.financeUseCases.summary(userId);
   427→        return {
   428→          budget: s.budget,
   429→          monthSpent: s.monthSpent,
   430→          remaining: s.remaining,
   431→          expectedRecurringRestOfMonth: s.expectedRecurringRestOfMonth,
   432→          dailyAverage: Math.round(s.dailyAverage),
   433→          categoryLimits: s.categoryLimits,
   434→        };
   435→      }
   436→      case 'TODAY_SCHEDULE': {
   437→        const day = dayKeyOf(now);
   438→        const occurrences = await this.dayOccurrences(userId, day);
   439→        return { day, items: occurrences.map((o) => ({ time: fmtHHMM(new Date(o.startAt)), title: o.title, kind: o.kind, status: o.status ?? null })) };
   440→      }
   441→      case 'TASKS_STATUS': {
   442→        const all = await this.tasks.listAll(userId);
   443→        const open = all.filter((t) => t.status === 'TODO' || t.status === 'IN_PROGRESS');
   444→        const overdue = open.filter((t) => t.deadline && t.deadline < now);
   445→        const completedToday = all.filter((t) => t.completedAt && dayKeyOf(t.completedAt) === dayKeyOf(now)).length;
   446→        return {
   447→          activeTotal: open.length,
   448→          overdueCount: overdue.length,
   449→          overdueTitles: overdue.slice(0, 5).map((t) => t.title),
   450→          completedToday,
   451→          nextDeadlines: open
   452→            .filter((t) => t.deadline)
   453→            .sort((a, b) => (a.deadline?.getTime() ?? 0) - (b.deadline?.getTime() ?? 0))
   454→            .slice(0, 5)
   455→            .map((t) => ({ title: t.title, deadline: t.deadline ? dayKeyOf(t.deadline) : null })),
   456→        };
   457→      }
   458→      default: {
   459→        const [summary, dayOccurrences] = await Promise.all([
   460→          this.financeUseCases.summary(userId),
   461→          this.dayOccurrences(userId, dayKeyOf(now)),
   462→        ]);
   463→        const allTasks = await this.tasks.listAll(userId);
   464→        const open = allTasks.filter((t) => t.status === 'TODO' || t.status === 'IN_PROGRESS');
   465→        return {
   466→          today: dayKeyOf(now),
   467→          scheduleToday: dayOccurrences.slice(0, 8).map((o) => ({ time: fmtHHMM(new Date(o.startAt)), title: o.title })),
   468→          activeTasks: open.length,
   469→          overdueTasks: open.filter((t) => t.deadline && t.deadline < now).length,
   470→          monthSpent: summary.monthSpent,
   471→          budget: summary.budget,
   472→          remaining: summary.remaining,
   473→          spentToday: summary.spentToday,
   474→        };
   475→      }
   476→    }
   477→  }

   479→  private async dayOccurrences(userId: string, day: string): Promise<OccurrenceDTO[]> {
   480→    const [events, slots, tasks] = await Promise.all([
   481→      this.events.listAll(userId),
   482→      this.plans.listSlotsForDay(userId, parseWallIso(day), new Date(parseWallIso(day).getTime() + 86_400_000)),
   483→      this.tasks.listAll(userId),
   484→    ]);
   485→    const taskMap = new Map(tasks.map((t) => [t.id, t]));
   486→    const out: OccurrenceDTO[] = [];
   487→    for (const e of events) {
   488→      if (dayKeyOf(e.startAt) !== day && !e.recurrence) continue;
   489→      out.push({ key: `e-${e.id}`, kind: 'EVENT', refId: e.id, title: e.title, startAt: e.startAt.toISOString(), endAt: e.endAt?.toISOString() ?? null, isRecurring: Boolean(e.recurrence) });
   490→    }
   491→    for (const s of slots) {
   492→      const t = taskMap.get(s.taskId);
   493→      out.push({ key: `s-${s.id}`, kind: 'PLANNED_TASK', refId: s.id, taskId: s.taskId, title: t?.title ?? 'مهمة', startAt: s.startAt.toISOString(), endAt: s.endAt.toISOString(), status: s.status });
   494→    }
   495→    return out.sort((a, b) => a.startAt.localeCompare(b.startAt));
   496→  }

   555→    firstName: string,

   559→  ): string {

   565→      parts.push(`${opener} يا ${firstName}: ${executed.map((e) => e.summary).join('، وكمان ')}.`);

   571→      parts.push(`معلش، معرفتش أنفذ (${failed.join('، ')}) — جرب تاني.`);

   589→    const lines: string[] = [];

   591→    if (d.monthSpent !== undefined) lines.push(`صرفت الشهر ده حوالي ${Math.round(Number(d.monthSpent))} جنيه`);
   592→    if (d.remaining !== undefined && d.remaining !== null) lines.push(`والباقي من الميزانية ${Math.round(Number(d.remaining))} جنيه`);
   593→    if (d.spentToday !== undefined) lines.push(`النهارده صرفت ${Math.round(Number(d.spentToday))} جنيه`);
   594→    if (d.activeTotal !== undefined) lines.push(`عندك ${d.activeTotal} مهام شغالة`);
   595→    const rawItems = d.items as unknown;
   596→    if (Array.isArray(rawItems) && rawItems.length > 0) {
   597→      const items = rawItems as { time?: string; title: string }[];
   598→      lines.push(`جدولك: ${items.map((i) => `${i.time ? i.time + ' ' : ''}${i.title}`).join('، ')}`);
   599→    }
   600→    return lines.length > 0 ? lines.join('، ') + '.' : 'مفيش بيانات كفاية أجاوب منها لسه.';
   601→  }

   618→function synthesizeActionFromIntent(intent: string, message: string): AiAction[] | null {
   619→  const normalized = String(intent ?? '').toUpperCase();
   620→  switch (normalized) {
   621→    case 'QUERY':
   622→      return [{ type: 'QUERY', queryType: 'GENERAL', question: message }];
   623→    case 'CHITCHAT':
   624→      return [{ type: 'CHITCHAT', message }];
   625→    case 'PLAN_DAY':
   626→      return [{ type: 'PLAN_DAY', date: null }];
   627→    default:
   628→      return null;
   629→  }
   630→}

   654→function sanitizeTaskFields(raw: unknown): Record<string, unknown> {
   655→  const fields = (raw ?? {}) as Record<string, unknown>;
   656→  const out: Record<string, unknown> = {};
   657→  if (str(fields.title)) out.title = str(fields.title);
   658→  if (str(fields.priority)) out.priority = str(fields.priority);
   659→  if (str(fields.deadline)) out.deadline = str(fields.deadline);
   660→  if (num(fields.estimatedMinutes) !== undefined) out.estimatedMinutes = num(fields.estimatedMinutes);
   661→  if (fields.description !== undefined) out.description = str(fields.description) ?? null;
   662→  return out;
   663→}

   665→function describeFields(fields: Record<string, unknown>): string {
   666→  const labels: Record<string, string> = {
