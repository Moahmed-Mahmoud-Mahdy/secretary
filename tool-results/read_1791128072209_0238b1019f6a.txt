    65→/** Client-side occurrence expansion for events on a given day (wall-clock keys). */
    66→function eventOccursOn(ev: EventDTO, dayKey: string): boolean {
    67→  const startDay = ev.startAt.slice(0, 10);
    68→  if (dayKey < startDay) return false;
    69→  if (!ev.recurrence) return dayKey === startDay;
    70→  const dayMs = Date.UTC(
    71→    Number(dayKey.slice(0, 4)),
    72→    Number(dayKey.slice(5, 7)) - 1,
    73→    Number(dayKey.slice(8, 10))
    74→  );
    75→  const startMs = Date.UTC(
    76→    Number(startDay.slice(0, 4)),
    77→    Number(startDay.slice(5, 7)) - 1,
    78→    Number(startDay.slice(8, 10))
    79→  );
    80→  if (ev.recurrence === 'DAILY') return dayMs >= startMs;
    81→  if (ev.recurrence === 'WEEKLY') {
    82→    const DAY = 86_400_000;
    83→    return (dayMs - startMs) % (7 * DAY) === 0;
    84→  }
    85→  if (ev.recurrence === 'MONTHLY') return dayKey.slice(8, 10) === startDay.slice(8, 10);
    86→  return dayKey === startDay;
    87→}

    89→function eventToOccurrence(ev: EventDTO, dayKey: string): OccurrenceDTO {
    90→  return {
    91→    key: `event-${ev.id}-${dayKey}`,
    92→    kind: 'EVENT',
    93→    refId: ev.id,
    94→    eventId: ev.id,
    95→    title: ev.title,
    96→    startAt: ev.startAt,
    97→    endAt: ev.endAt,
    98→    eventType: ev.eventType,
    99→    isRecurring: Boolean(ev.recurrence),
   100→  };
   101→}

   119→/** Sunday-anchored week start for a wall-clock day key (Egyptian week). */
   120→function weekStartKey(key: string): string {
   121→  const ms = Date.UTC(Number(key.slice(0, 4)), Number(key.slice(5, 7)) - 1, Number(key.slice(8, 10)));
   122→  const offset = new Date(ms).getUTCDay(); // 0=Sunday
   123→  return addDaysKey(key, -offset);
   124→}

   126→/**
   127→ * Celebration burst on habit check-in (BRD §16) — amber/emerald palette,
   128→ * particle count scales with the streak so milestones feel bigger.
   129→ */
   130→function celebrateStreak(streak: number): void {
   131→  if (typeof window === 'undefined') return;
   132→  const intensity = Math.min(5, 1 + Math.floor(streak / 3));
   133→  const colors = ['#f59e0b', '#fbbf24', '#fde68a', '#10b981', '#6ee7b7'];
   134→  const base = { colors, ticks: 120, gravity: 0.9, scalar: 0.9, zIndex: 9999 };
   135→  void confetti({
   136→    ...base,
   137→    particleCount: 40 + intensity * 20,
   138→    spread: 65 + intensity * 8,
   139→    origin: { x: 0.25, y: 0.7 },
   140→    angle: 60,
   141→  });
   142→  void confetti({
   143→    ...base,
   144→    particleCount: 40 + intensity * 20,
   145→    spread: 65 + intensity * 8,
   146→    origin: { x: 0.75, y: 0.7 },
   147→    angle: 120,
   148→  });
   149→  if (streak >= 7) {
   150→    // milestone week+: a golden rain from the top
   151→    void confetti({ ...base, particleCount: 90, spread: 100, startVelocity: 35, origin: { x: 0.5, y: 0.15 } });
   152→  }
   153→}
